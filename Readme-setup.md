## create a folder and open visual studio code in it
ex: /Desktop/new_latest/ai-meeting

# initialize git 
git init

# add origin 
git remote add origin https://openforge.gov.in/plugins/git/ai-meeting-intelligence/AI_Meeting_Backend.git

# always take pull from main 
git pull origin main

# create your branch
git checkout -b <your_branch_name>

# check your branch
git branch

# check if any change in any code
git status

# add latest changes in your branch
git add .

# commit all your changes
git commit -m "give_description_meaningfull_what_you_have_changed"

# push your changes
git push origin <you_branch>
